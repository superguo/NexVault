```kotlin
package com.nexvault.wallet.data.repository

import javax.inject.Inject
import javax.inject.Singleton

/**
 * Implementation of [TransactionRepository].
 *
 * For this prompt, only implements getRecentTransactionsForToken and
 * refreshTransactionHistory. Send/gas methods will be added in Prompt 16.
 */
@Singleton
class TransactionRepositoryImpl @Inject constructor(
    private val transactionDao: TransactionDao,
    private val blockExplorerApi: BlockExplorerApi,       // check actual class name
    private val blockExplorerApiFactory: Any?,             // check if factory pattern is used
    private val chainRepository: ChainRepository,
) : TransactionRepository {

    override suspend fun getRecentTransactionsForToken(
        chainId: Int,
        address: String,
        tokenContractAddress: String?,
        limit: Int,
    ): List<Transaction> {
        val entities = if (tokenContractAddress == null || tokenContractAddress == "native") {
            // For native coin: get transactions where tokenContractAddress is null
            transactionDao.getRecentNativeTransactions(
                chainId = chainId,
                address = address.lowercase(),
                limit = limit,
            )
        } else {
            // For ERC-20: filter by token contract address
            transactionDao.getRecentTokenTransactions(
                chainId = chainId,
                address = address.lowercase(),
                tokenContractAddress = tokenContractAddress.lowercase(),
                limit = limit,
            )
        }
        return entities.map { it.toDomain(address) }
    }

    override suspend fun refreshTransactionHistory(chainId: Int, address: String) {
        val chain = chainRepository.getChainById(chainId) ?: return
        try {
            // Fetch normal transactions
            val txResponse = blockExplorerApi.getTransactions(
                address = address,
                apiKey = chain.explorerApiKey, // check field name
            )
            val txEntities = txResponse.result?.map { it.toEntity(chainId) } ?: emptyList()

            // Fetch token transfers
            val tokenTxResponse = blockExplorerApi.getTokenTransfers(
                address = address,
                apiKey = chain.explorerApiKey,
            )
            val tokenTxEntities = tokenTxResponse.result?.map { it.toEntity(chainId) } ?: emptyList()

            // Upsert all
            transactionDao.upsertTransactions(txEntities + tokenTxEntities)
        } catch (e: Exception) {
            // Silently fail — cached data will be used
        }
    }
}
```

```kotlin
class GetTokenDetailUseCase @Inject constructor(
    private val tokenRepository: TokenRepository,
    private val walletRepository: WalletRepository,
    private val chainRepository: ChainRepository,

    operator fun invoke(contractAddress: String, chainId: Int): Flow<Token?> {

        return walletRepository.getCurrentWallet().flatMapLatest { wallet ->
            if (wallet != null) {
                tokenRepository.getToken(chainId, contractAddress, wallet.address)
            } else {
                kotlinx.coroutines.flow.flowOf(null)
            }
        }
    }
}
```

```kotlin
package com.nexvault.wallet.feature.tokens

import androidx.lifecycle.SavedStateHandle
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*

import javax.inject.Inject

/**
 * ViewModel for the Token Detail screen.
 *
 * Receives contractAddress and chainId via navigation arguments
 * through SavedStateHandle.
 *
 * Ref: doc/05-IMPLEMENTATION-PLAN-PHASE2.md Task 2.5.1
 */
@HiltViewModel
class TokenDetailViewModel @Inject constructor(
    savedStateHandle: SavedStateHandle,
    private val getTokenDetailUseCase: GetTokenDetailUseCase,
    private val getPriceHistoryUseCase: GetPriceHistoryUseCase,
    private val getRecentTokenTransactionsUseCase: GetRecentTokenTransactionsUseCase,
    private val refreshTransactionHistoryUseCase: RefreshTransactionHistoryUseCase,
) : ViewModel() {

    /** Navigation argument: the token's contract address (or "native"). */
    private val contractAddress: String = checkNotNull(
        savedStateHandle.get<String>("contractAddress")
    )

    /** Navigation argument: the chain ID. */
    private val chainId: Int = checkNotNull(
        savedStateHandle.get<Int>("chainId")
    )

    private val _uiState = MutableStateFlow(TokenDetailUiState())
    val uiState: StateFlow<TokenDetailUiState> = _uiState.asStateFlow()

    private val _selectedChartDays = MutableStateFlow(7)

    init {
        observeToken()
        loadChartData()
        loadRecentTransactions()
    }

    private fun observeToken() {
        viewModelScope.launch {
            getTokenDetailUseCase(contractAddress, chainId).collect { token ->
                _uiState.update { state ->
                    state.copy(
                        token = token,
                        isLoading = false,
                    )
                }
            }
        }
    }

    private fun loadChartData() {
        viewModelScope.launch {
            val token = _uiState.value.token
                ?: getTokenDetailUseCase(contractAddress, chainId).first()
            val coinGeckoId = token?.coinGeckoId ?: return@launch

            _uiState.update { it.copy(isChartLoading = true) }

            val result = getPriceHistoryUseCase(
                coinGeckoId = coinGeckoId,
                days = _selectedChartDays.value,
            )
            result.onSuccess { points ->
                _uiState.update { it.copy(
                    chartData = points,
                    selectedChartDays = _selectedChartDays.value,
                    isChartLoading = false,
                ) }
            }.onFailure {
                _uiState.update { it.copy(isChartLoading = false) }
            }
        }
    }

    private fun loadRecentTransactions() {
        viewModelScope.launch {
            try {
                // First try to refresh from network
                refreshTransactionHistoryUseCase(chainId)
            } catch (_: Exception) {
                // Use cached data
            }

            val transactions = getRecentTokenTransactionsUseCase(
                chainId = chainId,
                tokenContractAddress = contractAddress,
                limit = 5,
            )
            _uiState.update { it.copy(recentTransactions = transactions) }
        }
    }

    /**
     * Called when user selects a chart time range.
     */
    fun onChartRangeSelected(days: Int) {
        _selectedChartDays.value = days
        loadChartData()
    }

    /**
     * Called when user pulls to refresh.
     */
    fun onRefresh() {
        viewModelScope.launch {
            _uiState.update { it.copy(isRefreshing = true) }
            try {
                refreshTransactionHistoryUseCase(chainId)
                loadRecentTransactions()
            } catch (_: Exception) {}
            _uiState.update { it.copy(isRefreshing = false) }
        }
    }
}
```

```kotlin
    val isFailed = transaction.status == 2
    val isPending = transaction.status == 0

        isSwap -> Icons.Default.SwapHoriz

        isSwap -> Color(0xFF2196F3) // blue
        isSend -> MaterialTheme.colorScheme.error
        else -> Color(0xFF4CAF50) // green
    }

    val amountPrefix = when {
        isSend -> "-"
        isSwap -> ""
        else -> "+"
    }

    Row(
        modifier = modifier
            .fillMaxWidth()
            .padding(horizontal = 16.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        // Direction icon
        Box(
            modifier = Modifier
                .size(40.dp)
                .clip(CircleShape)
                .background(iconTint.copy(alpha = 0.12f)),
            contentAlignment = Alignment.Center,
        ) {
            Icon(
                imageVector = icon,
                contentDescription = transaction.type,
                tint = iconTint,
                modifier = Modifier.size(20.dp),
            )
        }

        Spacer(modifier = Modifier.width(12.dp))

        // Type and address
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = when {
                    isFailed -> "Failed"
                    isSwap -> "Swap"
                    isSend -> "Sent"
                    else -> "Received"
                },
                style = MaterialTheme.typography.bodyLarge,
                fontWeight = FontWeight.Medium,
            )
            Text(
                text = if (isSend) {
                    "To: ${truncateAddress(transaction.toAddress)}"
                } else {
                    "From: ${truncateAddress(transaction.fromAddress)}"
                },
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }

        // Amount and status
        Column(horizontalAlignment = Alignment.End) {
            val displayValue = formatTransactionValue(transaction.value, tokenDecimals)
            Text(
                text = "$amountPrefix$displayValue $tokenSymbol",
                style = MaterialTheme.typography.bodyMedium,
                fontWeight = FontWeight.Medium,
                color = if (isFailed) MaterialTheme.colorScheme.onSurfaceVariant else iconTint,
            )
            Text(
                text = when {
                    isPending -> "Pending"
                    isFailed -> "Failed"
                    else -> formatTimestamp(transaction.timestamp)
                },
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }
```

```kotlin
// --- Formatting helpers ---

/**
 * Truncates an Ethereum address to "0x1234...abcd" format.
 */
private fun truncateAddress(address: String): String {
    if (address.length < 10) return address
    return "${address.take(6)}...${address.takeLast(4)}"
}

/**
 * Formats a transaction value for display.
 */
private fun formatTransactionValue(value: java.math.BigDecimal, decimals: Int): String {
    val scaled = if (value.scale() > 6) {
        value.setScale(6, java.math.RoundingMode.DOWN)
    } else {
        value
    }
    return scaled.stripTrailingZeros().toPlainString()
}

/**
 * Formats a Unix timestamp (seconds) to a human-readable date string.
 */
private fun formatTimestamp(timestampSeconds: Long): String {
    val sdf = java.text.SimpleDateFormat("MMM dd", java.util.Locale.getDefault())
    return sdf.format(java.util.Date(timestampSeconds * 1000))
}

/**
 * Formats a fiat value like "$1,234.56".
 */
private fun formatFiatValue(value: Double): String {
    return "$${String.format("%,.2f", value)}"
}

/**
 * Formats a token balance for display, trimming excess decimals.
 */
private fun formatTokenBalance(balance: java.math.BigDecimal): String {
    return if (balance.scale() > 6) {
        balance.setScale(6, java.math.RoundingMode.DOWN).stripTrailingZeros().toPlainString()
    } else {
        balance.stripTrailingZeros().toPlainString()
    }
}
```
