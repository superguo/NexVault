### 1. Domain Layer — ChainRepository Interface & Use Cases

Create the repository interface and use cases in the `domain` module.

```kotlin
package com.nexvault.wallet.domain.repository

import com.nexvault.wallet.domain.model.Chain
import kotlinx.coroutines.flow.Flow

/**
 * Repository for managing blockchain network chains.
 *
 * Provides access to supported chains, the currently selected chain,
 * and the ability to switch between chains. The selected chain persists
 * across app restarts via DataStore.
 *
 * Ref: doc/05-IMPLEMENTATION-PLAN-PHASE2.md Task 2.3.1
 */
interface ChainRepository {

    /**
     * Observes the list of all supported chains.
     *
     * The list is hardcoded for now but could be dynamic in the future
     * (e.g., user-added custom networks).
     *
     * @return Flow emitting the list of supported chains
     */
    fun getSupportedChains(): Flow<List<Chain>>

    /**
     * Observes the currently selected chain.
     *
     * The selected chain determines which blockchain network is used
     * for all operations: balance fetching, transaction history,
     * sending transactions, NFT loading, etc.
     *
     * @return Flow emitting the currently selected chain
     */
    fun getSelectedChain(): Flow<Chain>

    /**
     * Changes the selected chain.
     *
     * This triggers a cascade of data refreshes across the app:
     * token balances, transaction history, NFT gallery, etc.
     *
     * @param chainId The chain ID to select (must be a supported chain)
     * @throws IllegalArgumentException if chainId is not supported
     */
    suspend fun setSelectedChain(chainId: Int)

    /**
     * Gets a chain by its ID synchronously.
     *
     * Useful for one-off lookups where a Flow is not needed.
     *
     * @param chainId The chain ID to look up
     * @return The Chain, or null if not supported
     */
    fun getChainById(chainId: Int): Chain?
}
```

```kotlin
package com.nexvault.wallet.data.repository

import com.nexvault.wallet.domain.model.Chain
import com.nexvault.wallet.domain.repository.ChainRepository
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.map
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Implementation of [ChainRepository] using a hardcoded chain list and
 * DataStore for persisting the selected chain.
 *
 * The chain list is currently static but structured to support future
 * dynamic chains (user-added custom networks).
 *
 * Ref: doc/05-IMPLEMENTATION-PLAN-PHASE2.md Task 2.3.2
 */
@Singleton
class ChainRepositoryImpl @Inject constructor(
    private val preferences: NexVaultPreferences, // from core-datastore
) : ChainRepository {

    /**
     * Hardcoded list of supported chains.
     *
     * Chain data matches doc/01-PROJECT-OVERVIEW.md Section 3.
     * RPC URLs use Infura/Alchemy with API key from BuildConfig for
     * Ethereum networks, and public RPCs for BSC/Polygon.
     *
     * NOTE: In a real production app, the Infura/Alchemy API key should
     * be injected here rather than hardcoded. For this portfolio project,
     * we use public endpoints or BuildConfig fields.
     */
    private val supportedChains: List<Chain> = listOf(
        Chain(
            chainId = 1,
            name = "Ethereum",
            symbol = "ETH",
            rpcUrl = "https://mainnet.infura.io/v3/", // append API key at runtime
            explorerUrl = "https://etherscan.io",
            explorerApiUrl = "https://api.etherscan.io",
            iconRes = 0, // placeholder — will be replaced with actual drawable res
            isTestnet = false,
            nativeCoinName = "Ethereum",
            nativeCoinCoinGeckoId = "ethereum",
        ),
        Chain(
            chainId = 11155111,
            name = "Sepolia Testnet",
            symbol = "ETH",
            rpcUrl = "https://sepolia.infura.io/v3/", // append API key at runtime
            explorerUrl = "https://sepolia.etherscan.io",
            explorerApiUrl = "https://api-sepolia.etherscan.io",
            iconRes = 0,
            isTestnet = true,
            nativeCoinName = "Sepolia ETH",
            nativeCoinCoinGeckoId = "ethereum", // uses ETH price for display
        ),
        Chain(
            chainId = 56,
            name = "BNB Smart Chain",
            symbol = "BNB",
            rpcUrl = "https://bsc-dataseed1.binance.org",
            explorerUrl = "https://bscscan.com",
            explorerApiUrl = "https://api.bscscan.com",
            iconRes = 0,
            isTestnet = false,
            nativeCoinName = "BNB",
            nativeCoinCoinGeckoId = "binancecoin",
        ),
        Chain(
            chainId = 137,
            name = "Polygon",
            symbol = "MATIC",
            rpcUrl = "https://polygon-rpc.com",
            explorerUrl = "https://polygonscan.com",
            explorerApiUrl = "https://api.polygonscan.com",
            iconRes = 0,
            isTestnet = false,
            nativeCoinName = "MATIC",
            nativeCoinCoinGeckoId = "matic-network",
        ),
    )

    /** Map for O(1) chain lookup by ID. */
    private val chainMap: Map<Int, Chain> = supportedChains.associateBy { it.chainId }

    /** Default chain if none is selected. */
    private val defaultChain: Chain = supportedChains.first()

    override fun getSupportedChains(): Flow<List<Chain>> {
        // Currently static, but returning Flow allows future dynamic chains
        return MutableStateFlow(supportedChains)
    }

    override fun getSelectedChain(): Flow<Chain> {
        return preferences.selectedChainId.map { chainId ->
            chainMap[chainId] ?: defaultChain
        }
    }

    override suspend fun setSelectedChain(chainId: Int) {
        require(chainMap.containsKey(chainId)) {
            "Unsupported chain ID: $chainId. Supported: ${chainMap.keys}"
        }
        preferences.setSelectedChainId(chainId)
    }

    override fun getChainById(chainId: Int): Chain? {
        return chainMap[chainId]
    }
}
```

### 6. ChainSelectorDropdown Composable (core-ui)

Create a dropdown composable for selecting the active chain. This appears in the
Home screen top bar.

```kotlin
package com.nexvault.wallet.core.ui.components

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.tooling.preview.Preview
import androidx.compose.ui.unit.dp

/**
 * Dropdown selector for switching between blockchain networks.
 *
 * Displays the currently selected chain with its icon and name.
 * Tapping opens a dropdown menu showing all supported chains.
 *
 * Ref: doc/03-UI-UX-DESIGN.md Section 2.3 — ChainSelector in Home Screen top bar
 * Ref: doc/05-IMPLEMENTATION-PLAN-PHASE2.md Task 2.3.3
 *
 * @param selectedChain The currently selected chain
 * @param supportedChains List of all available chains
 * @param onChainSelected Callback when user selects a different chain
 * @param modifier Optional modifier
 */
@Composable
fun ChainSelectorDropdown(
    selectedChain: ChainUi,
    supportedChains: List<ChainUi>,
    onChainSelected: (Int) -> Unit,
    modifier: Modifier = Modifier,
) {
    var expanded by remember { mutableStateOf(false) }

    Box(modifier = modifier) {
        // Selected chain display (clickable to open dropdown)
        Row(
            verticalAlignment = Alignment.CenterVertically,
            modifier = Modifier
                .clickable { expanded = true }
                .padding(horizontal = 12.dp, vertical = 8.dp),
        ) {
            // Chain icon
            Icon(
                painter = painterResource(id = selectedChain.iconRes),
                contentDescription = selectedChain.name,
                modifier = Modifier.size(24.dp),
                tint = Color.Unspecified, // preserve original icon colors
            )
            Spacer(modifier = Modifier.width(8.dp))
            // Chain name
            Text(
                text = selectedChain.name,
                style = MaterialTheme.typography.titleSmall,
                color = MaterialTheme.colorScheme.onSurface,
            )
            Spacer(modifier = Modifier.width(4.dp))
            // Dropdown arrow
            Icon(
                imageVector = Icons.Default.ArrowDropDown,
                contentDescription = "Select network",
                tint = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }

        // Dropdown menu
        DropdownMenu(
            expanded = expanded,
            onDismissRequest = { expanded = false },
        ) {
            supportedChains.forEach { chain ->
                DropdownMenuItem(
                    text = {
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Icon(
                                painter = painterResource(id = chain.iconRes),
                                contentDescription = chain.name,
                                modifier = Modifier.size(20.dp),
                                tint = Color.Unspecified,
                            )
                            Spacer(modifier = Modifier.width(8.dp))
                            Column {
                                Text(
                                    text = chain.name,
                                    style = MaterialTheme.typography.bodyMedium,
                                )
                                if (chain.isTestnet) {
                                    Text(
                                        text = "Testnet",
                                        style = MaterialTheme.typography.labelSmall,
                                        color = MaterialTheme.colorScheme.tertiary,
                                    )
                                }
                            }
                        }
                    },
                    onClick = {
                        expanded = false
                        onChainSelected(chain.chainId)
                    },
                    trailingIcon = if (chain.chainId == selectedChain.chainId) {
                        {
                            Icon(
                                imageVector = Icons.Default.Check,
                                contentDescription = "Selected",
                                tint = MaterialTheme.colorScheme.primary,
                            )
                        }
                    } else null,
                )
            }
        }
    }
}

/**
 * UI model for chain display in the selector.
 *
 * This is a presentation-layer model to avoid leaking domain models
 * into the composable. Map from domain [Chain] to [ChainUi] in the ViewModel.
 */
data class ChainUi(
    val chainId: Int,
    val name: String,
```

### 7. ChainIconMapper Utility (core-ui)

```kotlin

/**
 * Maps chain IDs to their corresponding drawable resource IDs.
 *
 * This keeps the chain icon resolution in the UI layer, avoiding
 * a dependency from the data layer on Android resources.
 */
object ChainIconMapper {

    /**
     * Returns the drawable resource ID for a chain's icon.
     *
     * @param chainId The EIP-155 chain ID
     * @return Drawable resource ID, or a default icon for unknown chains

    fun getIconRes(chainId: Int): Int {
        return when (chainId) {
            1 -> R.drawable.ic_chain_ethereum
            11155111 -> R.drawable.ic_chain_sepolia
            56 -> R.drawable.ic_chain_bsc
            137 -> R.drawable.ic_chain_polygon
            else -> R.drawable.ic_chain_ethereum // fallback to Ethereum icon
        }
    }
}
```

### 12. Common Issues to Watch For:

**Issue: `selectedChainId` Flow not found in NexVaultPreferences**
Check the exact property name. It might be `selectedChainId`, `selectedChain`,
`chainId`, or accessed via a method. Look at the actual implementation file.

**Issue: ChainConfigProvider conflict with ChainRepositoryImpl**
If both define chain data, choose one as the source of truth. The repository is the
better location since it's the domain/data boundary. The network config provider can
then query the repository (or they share a common chain definition object).

**Issue: R.drawable not resolving for chain icons**
Vector drawables must be valid XML. If the paths are malformed, they won't compile.
Simplify the paths if needed — even a plain colored circle works as a placeholder.
If all else fails, skip the vector drawables and use `0` for iconRes, handling the
fallback in the composable with a `Box` and colored `Surface`.

**Issue: core-ui depending on domain**
This is architecturally acceptable for mapper extensions. The UI layer can depend
on the domain layer. The domain layer MUST NOT depend on the UI layer.
Add `implementation(project(":domain"))` to `core-ui/build.gradle.kts`.

**Issue: DropdownMenu not positioning correctly**
`DropdownMenu` must be inside a `Box` (or similar container) that acts as the anchor.
The `Box` wrapping is included in the code above.

**Issue: Icon tint overriding vector colors**
If `tint = Color.Unspecified` doesn't work, try removing the `tint` parameter entirely.
In some Compose versions, you need to use `Image` instead of `Icon` to preserve
original vector colors. Alternatively, set `tint` to `Color.Unspecified` and verify
the specific Compose version handles it.
